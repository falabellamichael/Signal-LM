  <script>
    (function () {
      const SETTINGS_KEY = 'lmStudioLite.settings.v1';
      const media = window.matchMedia ? window.matchMedia('(prefers-color-scheme: dark)') : null;

      function readSettings() {
        try { return JSON.parse(localStorage.getItem(SETTINGS_KEY) || '{}') || {}; }
        catch { return {}; }
      }

      function writeSettings(next) {
        localStorage.setItem(SETTINGS_KEY, JSON.stringify(next || {}));
      }

      function resolveTheme(preference) {
        const pref = preference || readSettings().theme || 'system';
        if (pref === 'dark' || pref === 'light') return pref;
        return media && media.matches ? 'dark' : 'light';
      }

      function applyTheme(preference) {
        const settings = readSettings();
        const pref = preference || settings.theme || 'system';
        const resolved = resolveTheme(pref);
        document.documentElement.dataset.themePreference = resolved;
        
        // Apply gradient based on theme for professional polish
        if (resolved === 'light') {
          document.body.style.backgroundImage = "linear-gradient(135deg, #f8fafc 0%, #ffffff 100%)";
        } else {
          document.body.style.backgroundImage = "none";
        }
      }

      function applyThemeFromMedia() {
        const pref = readSettings().theme;
        if (!pref) return applyTheme(null); // use system default
        return applyTheme(pref);
      }

      window.LmStudioLiteTheme = {
        setTheme(theme) {
          writeSettings({ theme });
          setTimeout(() => applyTheme(theme), 10);
          showToast(`Set theme to ${theme}`);
        }
      };

      function setupTheme() {
        const pref = readSettings().theme || 'system';
        applyTheme(pref).catch(console.error);
      }

      // Apply system preference on initial load if no setting exists
      setupTheme();
    })();
  </script>

<script>
// Message formatting (minimal markdown support)
function formatMarkdown(text, theme = 'dark') {
  // Basic inline styling - more could be added later
  return text.replace(/\*\*(.*?)\*\*/g, '<b>$1</b>');
}

async function sendMessage() {
  const inputEl = document.getElementById('user-input');
  let message = inputEl.value.trim();
  
  if (!message) return;
  
  // Add user message
  appendMessage({
    id: Date.now(),
    text: formatMarkdown(message),
    isUser: true,
    avatarInitial: 'Me'
  });
  
  const originalText = inputEl.value;
  inputEl.value = '';
  scrollToBottom();
  await saveMessages();
  window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });

  // Stream response with typing animation effect
  try {
    const nativeBridge = window.lmStudioLiteNative || 
                        window.NativeInferenceBridge || 
                        window.AndroidInferenceBridge;
    
    if (nativeBridge && typeof nativeBridge.fetchCompletion === 'function') {
      const loadingIndicator = document.createElement('div');
      loadingIndicator.className = 'message-bubble user';
      loadingIndicator.style.justifyContent = 'flex-end';
      loadingIndicator.innerHTML = `<span style="display:inline-block;width:24px;height:24px;border-radius:50%;border-top:3px solid #fff;border-right:3px solid #fff;border-bottom:3px solid #fff;"></span>`;
      
      appendMessage({
        id: Date.now() + 1,
        text: '',
        isUser: false,
        avatarInitial: '🤖',
        loadingIndicator: loadingIndicator,
        loadingMessage: message, // store original prompt for copy/delete reference
      });
      
      scrollToBottom();

      const result = await nativeBridge.fetchCompletion({ 
        messages: [{ role: "user", content: JSON.stringify(message) }], 
        maxTokens: 1024 
      }, true);
      
      if (result?.text || result?.output_text) {
        const completionText = (result.text || result.output_text).trim();
        appendMessage({
          id: Date.now() + 2,
          text: formatMarkdown(completionText),
          isUser: false,
          avatarInitial: '🤖',
          loadingIndicator: null,
          copyText: originalText // for edit features in future
        }, () => {
          inputEl.focus();
        });
      } else if (result?.status) {
        appendMessage({
          id: Date.now() + 2,
          text: `Error: ${JSON.stringify(result)}`,
          isUser: false,
          avatarInitial: '❌',
          loadingIndicator: null
        });
      }

    } else {
      appendMessage({
        id: Date.now() + 1,
        text: "ℹ️ Your local inference bridge isn't ready. Make sure LM Studio Lite is running and the chat URL is set.",
        isUser: false,
        avatarInitial: '⚠️',
        loadingIndicator: null
      }, () => {
        inputEl.focus();
      });
    }

  } catch (error) {
    appendMessage({
      id: Date.now() + 1,
      text: `Connection failed: ${error.message || error}`,
      isUser: false,
      avatarInitial: '❌',
      loadingIndicator: null
    });
    console.error(error);
  }

  updateScrollStatus();
}

function appendMessage(msgData, callback) {
  const listEl = document.getElementById('messages-list');
  const bubbleEl = document.createElement('div');
  bubbleEl.className = `message-bubble ${msgData.isUser ? 'user' : 'chat'}`;
  
  let innerHtml = '';
  if (msgData.loadingIndicator) {
    innerHtml += `<div class="loading"></div>`; // animation element
    innerHtml += msgData.avatarInitial === '🤖' ? '<span class="bubble-text">Thinking...</span>' : '<span class="bubble-text"></span>';
  } else {
    innerHtml = `<div class="bubble-text">${msgData.text}</div>`;
  }

  if (msgData.loadingMessage) {
    // Add reference to original message for potential edit/delete feature
    bubbleEl.innerHTML += `<button class="btn-attach" style="color:#aaa;font-size:14px;" onclick="editMessage(this)">Edit</button>
                          <button class="btn-attach" style="color:#6366f1;font-size:12px;" onclick="copyToClipboard('${msgData.text}')" title="Copy response">📋</button>`;
  } else if (!msgData.loadingIndicator) {
    bubbleEl.innerHTML += `<div class="user-controls-bubble" onclick="window.history.go(-1)" title="Go back to previous message">&lt;</div>`;
  }

  // If it's a chat message, the user bubble should show controls differently
  if (!msgData.isUser && msgData.loadingIndicator) {
    bubbleEl.innerHTML += `<span class="bubble-text" style="opacity:0.5;">Thinking...</span> <button class="btn-attach" style="color:#aaa;font-size:14px;" onclick="editMessage(this)">Edit</button>
                           <button class="btn-attach" style="color:#6366f1;font-size:12px;" onclick="copyToClipboard('${msgData.loadingMessage}')">📋</button>`;
  } else if (!msgData.isUser) {
    bubbleEl.innerHTML += `<span class="bubble-text">${msgData.text}</span> <div class="user-controls-bubble" title="Copy response">📋</div>`;
  }

  // Avatar styling
  bubbleEl.className = msgData.isUser 
      ? 'message-bubble user' + (msgData.avatarInitial !== 'Me' && msgData.avatarInitial !== '' ? ` avatar-initial-${escapeCss(msgData.avatarInitial)}` : '')
      : 'message-bubble chat';
  
  // Chat bubble text alignment and spacing
  if (msgData.isUser) {
    bubbleEl.className = msgData.loadingIndicator 
        ? 'message-bubble user' + (msgData.avatarInitial !== 'Me' && msgData.avatarInitial !== '' ? ` avatar-initial-${escapeCss(msgData.avatarInitial)}` : '')
        : 'message-bubble user';
  }

  listEl.appendChild(bubbleEl);

  if (!msgData.loadingIndicator && !callback) {
    scrollToBottom();
  }
}

function editMessage(btn) {
  // Placeholder for future edit functionality - copy original prompt back to input
  btn.parentElement.previousElementSibling.click(); // go back one message (the user's)
}

function copyToClipboard(text) {
  navigator.clipboard.writeText(text).then(() => showToast('Copied to clipboard!'));
}

// Scroll and resize utilities
function scrollToBottom() {
  const container = document.querySelector('.messages-container');
  if (container) {
    container.scrollTop = container.scrollHeight;
  }
}

window.updateScrollStatus = function() {
  const header = document.querySelector('header');
  window.scrollY > 10 ? header.classList.add('scrolled') : header.classList.remove('scrolled');
  
  // Also update footer container
  const container = document.querySelector('.app-container');
  container.style.paddingBottom = '4rem';
};

// Resize textarea on input
document.getElementById('user-input').addEventListener('input', function() {
  this.style.height = '';
  this.style.height = Math.max(this.scrollHeight, 30) + 'px';
});

function scrollToBottomSmooth() {
  const container = document.querySelector('.messages-container');
  if (!container) return;
  const offset = window.innerHeight + container.getBoundingClientRect().top - window.scrollY + 120;
  requestAnimationFrame(() => container.scrollTop = offset);
}

// Save messages to localStorage (future feature - basic version)
window.saveMessages = async function() {
  const listEl = document.getElementById('messages-list');
  if (!listEl) return;
  
  const messages = [];
  listEl.querySelectorAll('.message-bubble').forEach(el => {
    messages.push({
      id: parseInt(el.dataset.id),
      text: el.querySelector('.bubble-text')?.textContent,
      isUser: el.classList.contains('user'),
      timestamp: new Date().toISOString()
    });
  });

  localStorage.setItem('chat-messages', JSON.stringify(messages));
};
